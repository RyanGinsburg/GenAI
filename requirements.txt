# Cornell Club Matching Agent — Python dependencies

# Install into the project venv:

# Web API
fastapi
uvicorn
python-multipart  # required by FastAPI for Form(...)/UploadFile in POST /chat

# Scraping (scraper/), and Firecrawl REST calls in research_agent.py (raw
# requests.post/get against api.firecrawl.dev — not the firecrawl-py SDK,
# since its class/method names have changed across versions).
requests
beautifulsoup4  # still used by scraper/scrape_campusgroups.py only

# LLM extraction + reasoning
anthropic

# Config / secrets loading from .env
python-dotenv

# Resume parsing (PDF -> text)
pypdf

# Embedding storage + cosine similarity
numpy

# Local embedding model for club matching (Step 2, matching.py)
sentence-transformers

# Accounts: stateless login tokens (Authorization: Bearer <token>). Password
# hashing uses the stdlib (hashlib.pbkdf2_hmac) — no extra dependency there.
# SQLite (data/app.db) uses the stdlib sqlite3 module — no dependency either.
pyjwt

# Verifies Google Sign-In ID tokens server-side (google.oauth2.id_token.
# verify_oauth2_token) — only needs GOOGLE_CLIENT_ID as the expected
# audience, not the client secret. Password-reset email uses stdlib smtplib
# — no dependency needed there.
google-auth
